# Despesas Operacionais — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Criar o módulo `/despesas` para lançamento de despesas operacionais (por loja ou geral da empresa) com integração no módulo Financeiro.

**Architecture:** Nova tabela `despesas` no banco com `lojaId` nullable (null = empresa geral). Server Actions protegidas por papel. Página Server Component com filtros via searchParams + componentes client para dialog e tabela. Card de resumo adicionado ao Financeiro.

**Tech Stack:** Next.js App Router, Drizzle ORM, Supabase (Postgres), Sonner (toasts), Lucide, shadcn/ui (Card, Table, Dialog, Select), CurrencyInput.

## Global Constraints

- Apenas `admin_geral` e `admin_socio` acessam e lançam despesas
- `lojaId` nullable: null = custo geral da empresa
- Sem comprovante, sem aprovação — lançamento vale na hora
- Categorias fixas (hardcoded): Aluguel, Energia elétrica, Água, Internet, Combustível, Compra de veículos, Marketing, Salários, Impostos, Manutenção, Outros
- Formas de pagamento: PIX, Transferência, Boleto, Dinheiro, Cartão
- Bancos (select): Sicoob, Itaú, Bradesco, Caixa Econômica Federal, Nubank, Santander, BTG Pactual, Outro
- `revalidatePath('/despesas')` e `revalidatePath('/financeiro')` após toda mutation
- Padrão de moeda: `CurrencyInput` de `@/components/ui/currency-input`
- Toast via `sonner`

```ts
'use server'
import { getUsuarioLogado } from '@/lib/auth'
import { db } from '@/lib/db'
import { despesas } from '@/lib/db/schema'
import { eq } from 'drizzle-orm'
import { revalidatePath } from 'next/cache'

const ADMINS = ['admin_geral', 'admin_socio'] as const

type DespesaInput = {
  lojaId: string | null
  categoria: string
  descricao?: string
  valor: string
  data: string
  formaPagamento: string
  banco: string
}

function assertAdmin(papel: string) {
  if (!ADMINS.includes(papel as typeof ADMINS[number])) {
    throw new Error('Sem permissão')
  }
}

function parseValor(v: string) {
  const n = parseFloat(v)
  if (isNaN(n) || n <= 0) throw new Error('Valor inválido')
  return v
}

export async function criarDespesaAction(dados: DespesaInput) {
  const usuario = await getUsuarioLogado()
  assertAdmin(usuario.papel)
  parseValor(dados.valor)

  await db.insert(despesas).values({
    lojaId:         dados.lojaId || null,
    usuarioId:      usuario.id,
    categoria:      dados.categoria,
    descricao:      dados.descricao?.trim() || null,
    valor:          dados.valor,
    data:           dados.data,
    formaPagamento: dados.formaPagamento,
    banco:          dados.banco,
  })

  revalidatePath('/despesas')
  revalidatePath('/financeiro')
}

export async function editarDespesaAction(id: string, dados: DespesaInput) {
  const usuario = await getUsuarioLogado()
  assertAdmin(usuario.papel)
  parseValor(dados.valor)

  await db.update(despesas).set({
    lojaId:         dados.lojaId || null,
    categoria:      dados.categoria,
    descricao:      dados.descricao?.trim() || null,
    valor:          dados.valor,
    data:           dados.data,
    formaPagamento: dados.formaPagamento,
    banco:          dados.banco,
  }).where(eq(despesas.id, id))

  revalidatePath('/despesas')
  revalidatePath('/financeiro')
}

export async function excluirDespesaAction(id: string) {
  const usuario = await getUsuarioLogado()
  assertAdmin(usuario.papel)

  await db.delete(despesas).where(eq(despesas.id, id))

  revalidatePath('/despesas')
  revalidatePath('/financeiro')
}
```

```tsx
  categoria: string
  descricao: string | null
  valor: string
  data: string

const CATEGORIAS = [
  'Aluguel', 'Energia elétrica', 'Água', 'Internet', 'Combustível',

  'Sicoob', 'Itaú', 'Bradesco', 'Caixa Econômica Federal',

type Props = {
  lojas: Loja[]
  modo: 'criar' | 'editar'
  inicial?: DespesaRow
  open?: boolean
  onOpenChange?: (v: boolean) => void
}

export function DespesaDialog({ lojas, modo, inicial, open, onOpenChange }: Props) {
  const router = useRouter()
  const [loading, setLoading] = useState(false)

  const hoje = new Date().toISOString().split('T')[0]

  const [lojaId, setLojaId]               = useState(inicial?.lojaId ?? '')
  const [categoria, setCategoria]         = useState(inicial?.categoria ?? '')
  const [descricao, setDescricao]         = useState(inicial?.descricao ?? '')
  const [valor, setValor]                 = useState(inicial?.valor ?? '')
  const [data, setData]                   = useState(inicial?.data ?? hoje)
  const [formaPagamento, setFormaPagamento] = useState(inicial?.formaPagamento ?? '')
  const [banco, setBanco]                 = useState(inicial?.banco ?? '')

  function resetar() {
    setLojaId('')
    setCategoria('')
    setDescricao('')
    setValor('')
    setData(hoje)
    setFormaPagamento('')
    setBanco('')
  }

  async function handleSalvar() {
    if (!categoria || !valor || !data || !formaPagamento || !banco) {
      toast.error('Preencha todos os campos obrigatórios')
      return
    }
    setLoading(true)
    try {
      const dados = {
        lojaId:         lojaId || null,
        categoria,
        descricao:      descricao.trim() || undefined,
        valor,
        data,
        formaPagamento,
        banco,
      }
      if (modo === 'criar') {
        await criarDespesaAction(dados)
        toast.success('Despesa registrada!')
        resetar()
      } else {
        await editarDespesaAction(inicial!.id, dados)
        toast.success('Despesa atualizada!')
      }
      onOpenChange?.(false)
      router.refresh()
    } catch (e: unknown) {
      toast.error(e instanceof Error ? e.message : 'Erro ao salvar')
    } finally {
      setLoading(false)
    }
  }

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="sm:max-w-lg">
        <DialogHeader>
          <DialogTitle>{modo === 'criar' ? 'Nova Despesa' : 'Editar Despesa'}</DialogTitle>
        </DialogHeader>

        <div className="space-y-4 py-2">
          {/* Loja */}
          <div className="space-y-1.5">
            <Label className="text-xs">Loja</Label>
            <select
              value={lojaId}
              onChange={e => setLojaId(e.target.value)}
              className="h-9 w-full rounded-lg border border-input bg-transparent px-2.5 text-sm outline-none"
            >
              <option value="">Geral — empresa toda</option>
              {lojas.map(l => (
                <option key={l.id} value={l.id}>{l.nome}</option>
              ))}
            </select>
          </div>

          {/* Categoria */}
          <div className="space-y-1.5">
            <Label className="text-xs">Categoria *</Label>
            <select
              value={categoria}
              onChange={e => setCategoria(e.target.value)}
              className="h-9 w-full rounded-lg border border-input bg-transparent px-2.5 text-sm outline-none"
            >
              <option value="">Selecione...</option>
              {CATEGORIAS.map(c => (
                <option key={c} value={c}>{c}</option>
              ))}
            </select>
          </div>

          {/* Valor + Data */}
          <div className="grid grid-cols-2 gap-3">
            <div className="space-y-1.5">
              <Label className="text-xs">Valor (R$) *</Label>
              <CurrencyInput value={valor} onChange={setValor} placeholder="0,00" />
            </div>
            <div className="space-y-1.5">
              <Label className="text-xs">Data *</Label>
              <Input type="date" value={data} onChange={e => setData(e.target.value)} className="h-9 text-sm" />
            </div>
          </div>

          {/* Forma de pagamento + Banco */}
          <div className="grid grid-cols-2 gap-3">
            <div className="space-y-1.5">
              <Label className="text-xs">Forma de pagamento *</Label>
              <select
                value={formaPagamento}
                onChange={e => setFormaPagamento(e.target.value)}
                className="h-9 w-full rounded-lg border border-input bg-transparent px-2.5 text-sm outline-none"
              >
                <option value="">Selecione...</option>
                {FORMAS_PAGAMENTO.map(f => (
                  <option key={f} value={f}>{f}</option>
                ))}
              </select>
            </div>
            <div className="space-y-1.5">
```

- [ ] **Step 2: Criar despesas-table.tsx**

```tsx
'use client'
import { useState, useMemo } from 'react'
import { useRouter } from 'next/navigation'
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table'
import { Input } from '@/components/ui/input'
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
import { Button } from '@/components/ui/button'
import { Trash2 } from 'lucide-react'
import { toast } from 'sonner'
import { excluirDespesaAction } from '../actions'
import { EditarDespesaButton } from './despesa-dialog'

type Loja = { id: string; nome: string }

type DespesaRow = {
  id: string
  lojaId: string | null
  lojaNome: string | null
  categoria: string
  descricao: string | null
  valor: string
  data: string
  formaPagamento: string
  banco: string
  usuarioNome: string
}

type Props = { despesas: DespesaRow[]; lojas: Loja[] }

function fmt(v: string) {
  return Number(v).toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' })
}

function fmtData(d: string) {
  return new Date(d + 'T12:00:00').toLocaleDateString('pt-BR')
}

export function DespesasTable({ despesas, lojas }: Props) {
  const router = useRouter()
  const [excluindo, setExcluindo] = useState<string | null>(null)
  const [filtroCategoria, setFiltroCategoria] = useState('')
  const [filtroLoja, setFiltroLoja]           = useState('')
  const [filtroBusca, setFiltroBusca]         = useState('')

  const categorias = useMemo(() =>
    [...new Set(despesas.map(d => d.categoria))].sort(),
    [despesas]
  )

  const dados = useMemo(() => despesas.filter(d => {
    if (filtroCategoria && d.categoria !== filtroCategoria) return false
    if (filtroLoja === '__geral' && d.lojaId !== null) return false
    if (filtroLoja && filtroLoja !== '__geral' && d.lojaId !== filtroLoja) return false
    if (filtroBusca) {
      const q = filtroBusca.toLowerCase()
      if (!d.descricao?.toLowerCase().includes(q) &&
          !d.categoria.toLowerCase().includes(q) &&
          !d.banco.toLowerCase().includes(q)) return false
    }
    return true
  }), [despesas, filtroCategoria, filtroLoja, filtroBusca])

  const total = useMemo(() =>
    dados.reduce((s, d) => s + Number(d.valor), 0),
    [dados]
  )

  async function handleExcluir(id: string) {
    if (!confirm('Excluir esta despesa?')) return
    setExcluindo(id)
    try {
      await excluirDespesaAction(id)
      toast.success('Despesa excluída')
      router.refresh()
    } catch (e: unknown) {
      toast.error(e instanceof Error ? e.message : 'Erro ao excluir')
    } finally {
      setExcluindo(null)
    }
  }

  if (despesas.length === 0) {
    return <p className="text-sm text-slate-400 py-8 text-center">Nenhuma despesa encontrada.</p>
  }

  return (
    <div className="space-y-3">
      {/* Filtros inline */}
      <div className="flex gap-2 flex-wrap">
        <Select value={filtroCategoria || '__todas'} onValueChange={v => setFiltroCategoria(v === '__todas' ? '' : v)}>
          <SelectTrigger className="h-8 w-44 text-xs">
            <SelectValue placeholder="Categoria" />
          </SelectTrigger>
          <SelectContent>
            <SelectItem value="__todas">Todas as categorias</SelectItem>
            {categorias.map(c => <SelectItem key={c} value={c}>{c}</SelectItem>)}
          </SelectContent>
        </Select>

        <Select value={filtroLoja || '__todas'} onValueChange={v => setFiltroLoja(v === '__todas' ? '' : v)}>
          <SelectTrigger className="h-8 w-44 text-xs">
            <SelectValue placeholder="Loja" />
          </SelectTrigger>
          <SelectContent>
            <SelectItem value="__todas">Todas as lojas</SelectItem>
            <SelectItem value="__geral">Geral (empresa)</SelectItem>
            {lojas.map(l => <SelectItem key={l.id} value={l.id}>{l.nome}</SelectItem>)}
          </SelectContent>
        </Select>

        <Input
          className="h-8 text-xs w-48"
          placeholder="Buscar descrição / banco..."
          value={filtroBusca}
          onChange={e => setFiltroBusca(e.target.value)}
        />
      </div>

      <Table>
        <TableHeader>
          <TableRow>
            <TableHead>Data</TableHead>
            <TableHead>Categoria</TableHead>
            <TableHead>Descrição</TableHead>
            <TableHead>Loja</TableHead>
            <TableHead>Pgto</TableHead>
            <TableHead>Banco</TableHead>
            <TableHead className="text-right">Valor</TableHead>
            <TableHead className="w-16" />
          </TableRow>
        </TableHeader>
        <TableBody>
          {dados.length === 0 && (
            <TableRow>
              <TableCell colSpan={8} className="py-8 text-center text-sm text-slate-400">
                Nenhuma despesa para os filtros aplicados.
              </TableCell>
            </TableRow>
          )}
          {dados.map(d => (
            <TableRow key={d.id}>
              <TableCell className="whitespace-nowrap text-sm">{fmtData(d.data)}</TableCell>
              <TableCell className="text-sm">{d.categoria}</TableCell>
              <TableCell className="text-sm text-slate-500 max-w-[180px] truncate">{d.descricao ?? '—'}</TableCell>
              <TableCell className="text-sm">
                {d.lojaNome ?? <span className="text-slate-400 text-xs italic">Geral</span>}
              </TableCell>
              <TableCell className="text-sm">{d.formaPagamento}</TableCell>
              <TableCell className="text-sm">{d.banco}</TableCell>
              <TableCell className="text-right font-mono text-sm text-red-600">
                -{fmt(d.valor)}
              </TableCell>
              <TableCell>
                <div className="flex items-center gap-1">
                  <EditarDespesaButton lojas={lojas} despesa={d} />
                  <Button
                    variant="ghost"
                    size="icon"
                    className="h-7 w-7 text-red-400 hover:text-red-600"
                    onClick={() => handleExcluir(d.id)}
                    disabled={excluindo === d.id}
                  >
                    <Trash2 size={14} />
                  </Button>
                </div>
              </TableCell>
            </TableRow>
          ))}
          {dados.length > 0 && (
```

- [ ] **Step 2: Criar page.tsx**

```tsx
import { getUsuarioLogado } from '@/lib/auth'
import { db } from '@/lib/db'
import { despesas, lojas, usuarios } from '@/lib/db/schema'
import { eq, desc, and, gte, lte, sql } from 'drizzle-orm'
import { redirect } from 'next/navigation'
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
import { DespesasTable } from './components/despesas-table'
import { NovaDespesaButton } from './components/despesa-dialog'
import { MesSelector } from './components/mes-selector'

const ADMINS = ['admin_geral', 'admin_socio']

export default async function DespesasPage({
  searchParams,
}: {
  searchParams: Promise<Record<string, string | undefined>>
}) {
  const usuario = await getUsuarioLogado()
  if (!ADMINS.includes(usuario.papel)) redirect('/dashboard')

  const params = await searchParams
  const mesParam = params.mes ?? new Date().toISOString().slice(0, 7)
  const [ano, mes] = mesParam.split('-').map(Number)
  const inicioMes = `${mesParam}-01`
  const ultimoDia = new Date(ano, mes, 0).toISOString().split('T')[0]

  const [rows, lojasOpts, totaisPorLoja] = await Promise.all([
    db
      .select({
        id:             despesas.id,
        lojaId:         despesas.lojaId,
        lojaNome:       lojas.nome,
        categoria:      despesas.categoria,
        descricao:      despesas.descricao,
        valor:          despesas.valor,
        data:           despesas.data,
        formaPagamento: despesas.formaPagamento,
        banco:          despesas.banco,
        usuarioNome:    usuarios.nome,
      })
      .from(despesas)
      .leftJoin(lojas,    eq(despesas.lojaId,    lojas.id))
      .innerJoin(usuarios, eq(despesas.usuarioId, usuarios.id))
      .where(and(
        gte(despesas.data, inicioMes),
        lte(despesas.data, ultimoDia),
      ))
      .orderBy(desc(despesas.data), desc(despesas.criadoEm)),

    db.select({ id: lojas.id, nome: lojas.nome }).from(lojas).orderBy(lojas.nome),

    db
      .select({
        lojaId:   despesas.lojaId,
        lojaNome: lojas.nome,
        total:    sql<string>`COALESCE(SUM(${despesas.valor}), '0')`,
      })
      .from(despesas)
      .leftJoin(lojas, eq(despesas.lojaId, lojas.id))
      .where(and(
        gte(despesas.data, inicioMes),
        lte(despesas.data, ultimoDia),
      ))
      .groupBy(despesas.lojaId, lojas.nome),
  ])

  const totalGeral   = rows.reduce((s, r) => s + Number(r.valor), 0)
  const totalLojas   = totaisPorLoja.filter(t => t.lojaId !== null).reduce((s, t) => s + Number(t.total), 0)
  const totalEmpresa = totaisPorLoja.filter(t => t.lojaId === null).reduce((s, t) => s + Number(t.total), 0)

  function fmt(v: number) {
    return v.toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' })
  }

  const mesExibicao = new Date(`${mesParam}-15`).toLocaleDateString('pt-BR', { month: 'long', year: 'numeric' })

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-2xl font-bold">Despesas Operacionais</h1>
          <p className="text-sm text-slate-500 mt-0.5 capitalize">{mesExibicao}</p>
        </div>
        <div className="flex items-center gap-3">
          <MesSelector mesAtual={mesParam} />
          <NovaDespesaButton lojas={lojasOpts} />
        </div>
      </div>

      {/* Cards de resumo */}
      <div className="grid grid-cols-3 gap-4">
        <Card>
          <CardHeader className="pb-2"><CardTitle className="text-sm text-slate-500">Total do Período</CardTitle></CardHeader>
          <CardContent><p className="text-2xl font-bold text-red-600">{fmt(totalGeral)}</p></CardContent>
        </Card>
        <Card>
          <CardHeader className="pb-2"><CardTitle className="text-sm text-slate-500">Por Loja</CardTitle></CardHeader>
          <CardContent><p className="text-2xl font-bold text-orange-600">{fmt(totalLojas)}</p></CardContent>
        </Card>
        <Card>
          <CardHeader className="pb-2"><CardTitle className="text-sm text-slate-500">Geral (Empresa)</CardTitle></CardHeader>
          <CardContent><p className="text-2xl font-bold text-orange-600">{fmt(totalEmpresa)}</p></CardContent>
        </Card>
      </div>

      <Card>
        <CardHeader><CardTitle className="text-base">Lançamentos</CardTitle></CardHeader>
        <CardContent>
          <DespesasTable despesas={rows} lojas={lojasOpts} />
        </CardContent>
      </Card>
    </div>
  )
}
```

> **Nota:** O `input[type=month]` usa `onChange` com `window.location.href` — o mesmo padrão dos filtros simples já existentes no projeto (ex: `caixa-filtros.tsx`). Se preferir, pode fazer um componente Client separado para o seletor de mês, mas não é necessário aqui porque é o único controle.
