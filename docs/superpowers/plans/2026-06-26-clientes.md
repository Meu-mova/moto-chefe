# Módulo de Clientes — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Criar módulo de cadastro de clientes com importação de 2.016 registros do Bling, histórico por cliente e integração com vendas e ordens de serviço.

**Architecture:** Tabela `clientes` central com enums `tipo_pessoa` e `sexo`. Coluna `cliente_id` adicionada (nullable) em `vendas` e `ordens_servico`. Script one-time lê CSV Latin-1, classifica sexo via Claude API e insere no banco. Componente `ClienteBusca` reutilizável faz autocomplete com server action e é usado nos dialogs de venda e OS.

**Tech Stack:** Next.js 16 App Router, Drizzle ORM 0.45, Supabase Postgres, `@anthropic-ai/sdk`, shadcn/ui, sonner, lucide-react.

## Global Constraints

- Nunca usar `useEffect` para buscar dados — preferir server actions ou server components.
- Migrations aplicadas via Supabase MCP (`apply_migration`) — **não** via `npm run db:migrate` (journal está defasado).
- `onValueChange` do Select retorna `string | null` — sempre usar `v ?? ''`.
- `'use server'` em `actions.ts`, `'use client'` nos componentes interativos.
- Mecânico não acessa o módulo de clientes — redirecionar com `redirect('/dashboard')`.
- Exclusão de cliente restrita a `admin_geral`.
- CPF/CNPJ armazenado somente dígitos (sem máscara).
- `cliente_id` é nullable em vendas e OS — não quebra registros existentes.

```typescript
import fs from 'fs'
import Anthropic from '@anthropic-ai/sdk'
import { db } from '../src/lib/db'
import { clientes } from '../src/lib/db/schema'

const CSV_PATH = 'D:\\ECOMOV\\exportar clientes Bling.csv'

function parseCSV(content: string): Record<string, string>[] {
  const lines = content.split('\n').filter(l => l.trim())
  const [headerLine, ...rows] = lines
  const cols = headerLine.split(';').map(c => c.trim())
  return rows.map(row => {
    const values = row.split(';')
    return Object.fromEntries(cols.map((c, i) => [c, (values[i] ?? '').trim()]))
  })
}

function primeiroNome(nome: string): string {
  return nome.trim().split(/\s+/)[0] ?? ''
}

async function classificarSexo(
  nomes: string[]
): Promise<Record<string, 'masculino' | 'feminino' | 'nao_identificado'>> {
  const client = new Anthropic()
  const resultado: Record<string, 'masculino' | 'feminino' | 'nao_identificado'> = {}
  const BATCH = 80

  for (let i = 0; i < nomes.length; i += BATCH) {
    const lote = nomes.slice(i, i + BATCH)
    const resp = await client.messages.create({
      model: 'claude-haiku-4-5-20251001',
      max_tokens: 2048,
      messages: [{
        role: 'user',
        content: `Classifique o sexo de cada primeiro nome brasileiro abaixo como "masculino", "feminino" ou "nao_identificado".\nRetorne APENAS um objeto JSON, sem texto adicional, no formato: {"Nome": "masculino", ...}\n\nNomes:\n${lote.join('\n')}`,
      }],
    })

    const text = resp.content[0].type === 'text' ? resp.content[0].text : '{}'
    try {
      const match = text.match(/\{[\s\S]*\}/)
      const parsed = JSON.parse(match?.[0] ?? '{}') as Record<string, string>
      for (const [nome, sexo] of Object.entries(parsed)) {
        if (sexo === 'masculino' || sexo === 'feminino' || sexo === 'nao_identificado') {
          resultado[nome] = sexo
        } else {
          resultado[nome] = 'nao_identificado'
        }
      }
    } catch {
      lote.forEach(n => { resultado[n] = 'nao_identificado' })
    }

    const loteNum = Math.floor(i / BATCH) + 1
    const total = Math.ceil(nomes.length / BATCH)
    console.log(`Classificação: lote ${loteNum}/${total}`)
  }

  return resultado
}

async function main() {
  const content = fs.readFileSync(CSV_PATH, 'latin1')
  const rows = parseCSV(content).filter(r => r['Nome'])
  console.log(`Total de registros: ${rows.length}`)

  const nomesUnicos = [...new Set(rows.map(r => primeiroNome(r['Nome'])))]
  console.log(`Primeiros nomes únicos: ${nomesUnicos.length}`)

  const sexoMap = await classificarSexo(nomesUnicos)
  console.log('Classificação de sexo concluída.')

  const registros = rows.map(r => ({
    nome: r['Nome'],
    tipoPessoa: (r['Tipo pessoa']?.includes('ur') ? 'juridica' : 'fisica') as 'fisica' | 'juridica',
    cpfCnpj: r['CNPJ / CPF']?.replace(/\D/g, '') || null,
    sexo: (sexoMap[primeiroNome(r['Nome'])] ?? 'nao_identificado') as 'masculino' | 'feminino' | 'nao_identificado',
    email: r['E-mail'] || null,
    celular: r['Celular'] || null,
    endereco: r['Endereço'] || null,
    numero: r['Número'] || null,
    complemento: r['Complemento'] || null,
    bairro: r['Bairro'] || null,
    cep: r['CEP'] || null,
    cidade: r['Cidade'] || null,
    uf: r['UF'] || null,
  }))

  const BATCH_INSERT = 100
  let total = 0

  for (let i = 0; i < registros.length; i += BATCH_INSERT) {
    const lote = registros.slice(i, i + BATCH_INSERT)
    const comCpf = lote.filter(r => r.cpfCnpj)
    const semCpf = lote.filter(r => !r.cpfCnpj)

    if (comCpf.length > 0) {
      await db.insert(clientes).values(comCpf).onConflictDoNothing({ target: clientes.cpfCnpj })
    }
    if (semCpf.length > 0) {
      await db.insert(clientes).values(semCpf)
    }
    total += lote.length
    console.log(`Inseridos: ${total}/${registros.length}`)
  }

  console.log('✓ Importação concluída!')
}

main().catch(err => { console.error(err); process.exit(1) })
```

Criar `src/app/(dashboard)/clientes/actions.ts`:

```typescript
'use server'
import { db } from '@/lib/db'
import { clientes, vendas, ordensServico, motos, modelos } from '@/lib/db/schema'
import { eq, ilike, or, and, desc, SQL } from 'drizzle-orm'
import { getUsuarioLogado } from '@/lib/auth'
import { revalidatePath } from 'next/cache'

export type ClienteInput = {
  nome: string
  tipoPessoa: 'fisica' | 'juridica'
  cpfCnpj?: string | null
  sexo: 'masculino' | 'feminino' | 'nao_identificado'
  email?: string | null
  celular?: string | null
  endereco?: string | null
  numero?: string | null
  complemento?: string | null
  bairro?: string | null
  cep?: string | null
  cidade?: string | null
  uf?: string | null
}

function limparCpfCnpj(v?: string | null) {
  return v?.replace(/\D/g, '') || null
}

export async function listarClientesAction(params: {
  busca?: string
  tipoPessoa?: string
  sexo?: string
  cidade?: string
}) {
  const usuario = await getUsuarioLogado()
  if (usuario.papel === 'mecanico') throw new Error('Sem permissão')

  const conds: SQL[] = []
  if (params.busca) {
    conds.push(or(
      ilike(clientes.nome, `%${params.busca}%`),
      ilike(clientes.cpfCnpj, `%${params.busca.replace(/\D/g, '')}%`),
      ilike(clientes.celular, `%${params.busca}%`)
    )!)
  }
  if (params.tipoPessoa && params.tipoPessoa !== 'todos') {
    conds.push(eq(clientes.tipoPessoa, params.tipoPessoa as 'fisica' | 'juridica'))
  }
  if (params.sexo && params.sexo !== 'todos') {
    conds.push(eq(clientes.sexo, params.sexo as 'masculino' | 'feminino' | 'nao_identificado'))
  }
  if (params.cidade && params.cidade !== 'todos') {
    conds.push(eq(clientes.cidade, params.cidade))
  }

  return db.select().from(clientes)
    .where(conds.length > 0 ? and(...conds) : undefined)
    .orderBy(clientes.nome)
    .limit(200)
}

export async function buscarClientesAction(termo: string) {
  if (termo.length < 3) return []
  const termoCpf = termo.replace(/\D/g, '')
  return db.select({
    id: clientes.id,
    nome: clientes.nome,
    cpfCnpj: clientes.cpfCnpj,
    celular: clientes.celular,
  }).from(clientes)
    .where(or(
      ilike(clientes.nome, `%${termo}%`),
      termoCpf.length >= 3 ? ilike(clientes.cpfCnpj, `%${termoCpf}%`) : undefined,
    ))
    .orderBy(clientes.nome)
    .limit(5)
}

export async function buscarClientePorIdAction(id: string) {
  const [cliente] = await db.select().from(clientes).where(eq(clientes.id, id)).limit(1)
  return cliente ?? null
}

export async function listarCidadesAction() {
  const rows = await db.selectDistinct({ cidade: clientes.cidade })
    .from(clientes)
    .orderBy(clientes.cidade)
  return rows.map(r => r.cidade).filter(Boolean) as string[]
}

export async function criarClienteAction(dados: ClienteInput): Promise<string> {
  const usuario = await getUsuarioLogado()
  if (usuario.papel === 'mecanico') throw new Error('Sem permissão')

  const cpfCnpj = limparCpfCnpj(dados.cpfCnpj)

  try {
    const [novo] = await db.insert(clientes).values({
      ...dados,
      cpfCnpj,
    }).returning({ id: clientes.id })
    revalidatePath('/clientes')
    return novo.id
  } catch (e: unknown) {
    const msg = e instanceof Error ? e.message : ''
    if (msg.includes('cpf_cnpj')) throw new Error('CPF/CNPJ já cadastrado')
    throw e
  }
}

export async function atualizarClienteAction(id: string, dados: Partial<ClienteInput>) {
  const usuario = await getUsuarioLogado()
  if (usuario.papel === 'mecanico') throw new Error('Sem permissão')

  const cpfCnpj = dados.cpfCnpj !== undefined ? limparCpfCnpj(dados.cpfCnpj) : undefined

  try {
    await db.update(clientes).set({
      ...dados,
      ...(cpfCnpj !== undefined ? { cpfCnpj } : {}),
      atualizadoEm: new Date(),
    }).where(eq(clientes.id, id))
    revalidatePath('/clientes')
    revalidatePath(`/clientes/${id}`)
  } catch (e: unknown) {
    const msg = e instanceof Error ? e.message : ''
    if (msg.includes('cpf_cnpj')) throw new Error('CPF/CNPJ já cadastrado')
    throw e
  }
}

export async function excluirClienteAction(id: string) {
  const usuario = await getUsuarioLogado()
  if (usuario.papel !== 'admin_geral') throw new Error('Sem permissão')
  await db.delete(clientes).where(eq(clientes.id, id))
  revalidatePath('/clientes')
}

export async function buscarHistoricoClienteAction(id: string, cpfCnpj: string | null) {
  const condicaoVendas = cpfCnpj
    ? or(eq(vendas.clienteId, id), eq(vendas.clienteCpf, cpfCnpj))!
    : eq(vendas.clienteId, id)

  const [vendasCliente, osCliente] = await Promise.all([
    db.select({
      id: vendas.id,
      numero: vendas.numero,
      dataVenda: vendas.dataVenda,
      valor: vendas.valor,
      formaPagamento: vendas.formaPagamento,
      marca: modelos.marca,
      modelo: modelos.modelo,
      chassi: motos.chassi,
    })
      .from(vendas)
      .innerJoin(motos, eq(vendas.motoId, motos.id))
      .innerJoin(modelos, eq(motos.modeloId, modelos.id))
      .where(condicaoVendas)
      .orderBy(desc(vendas.dataVenda)),

    db.select({
      id: ordensServico.id,
      dataAbertura: ordensServico.dataAbertura,
      dataConclusao: ordensServico.dataConclusao,
      status: ordensServico.status,
      descricao: ordensServico.descricao,
      marca: modelos.marca,
      modelo: modelos.modelo,
      chassi: motos.chassi,
    })
      .from(ordensServico)
      .innerJoin(motos, eq(ordensServico.motoId, motos.id))
      .innerJoin(modelos, eq(motos.modeloId, modelos.id))
      .where(eq(ordensServico.clienteId, id))
      .orderBy(desc(ordensServico.dataAbertura)),
  ])

  return { vendas: vendasCliente, os: osCliente }
}
```

```typescript
  if (!v) return ''
  if (v.length === 11) return v.replace(/(\d{3})(\d{3})(\d{3})(\d{2})/, '$1.$2.$3-$4')
  if (v.length === 14) return v.replace(/(\d{2})(\d{3})(\d{3})(\d{4})(\d{2})/, '$1.$2.$3/$4-$5')
  return v
}

export function ClienteBusca({ onSelect, clienteSelecionado, onClear, label = 'Cliente' }: Props) {
  const [termo, setTermo] = useState('')
  const [resultados, setResultados] = useState<Resultado[]>([])
  const [buscando, setBuscando] = useState(false)
  const [aberto, setAberto] = useState(false)
  const [novoOpen, setNovoOpen] = useState(false)
  const [novoNome, setNovoNome] = useState('')
  const [novoCelular, setNovoCelular] = useState('')

  const [salvando, setSalvando] = useState(false)
  const debounce = useRef<ReturnType<typeof setTimeout>>()
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    function handleClick(e: MouseEvent) {
      if (containerRef.current && !containerRef.current.contains(e.target as Node)) {
        setAberto(false)
      }

            {resultados.map(r => (
              <button
                key={r.id}

              <Label>Nome *</Label>
              <Input value={novoNome} onChange={e => setNovoNome(e.target.value)} />
            </div>
            <div className="space-y-1">
              <Label>Tipo de Pessoa</Label>
              <Select value={novoTipo} onValueChange={v => setNovoTipo((v ?? 'fisica') as 'fisica' | 'juridica')}>
                <SelectTrigger><SelectValue /></SelectTrigger>
                <SelectContent>
                  <SelectItem value="fisica">Pessoa Física</SelectItem>
                  <SelectItem value="juridica">Pessoa Jurídica</SelectItem>
                </SelectContent>
              </Select>
            </div>
            <div className="space-y-1">
              <Label>CPF / CNPJ</Label>
              <Input value={novoCpf} onChange={e => setNovoCpf(e.target.value)} placeholder="Opcional" />
            </div>
            <div className="space-y-1">
              <Label>Celular</Label>
              <Input value={novoCelular} onChange={e => setNovoCelular(e.target.value)} placeholder="Opcional" />
            </div>
          </div>
          <DialogFooter>
            <Button variant="outline" onClick={() => setNovoOpen(false)}>Cancelar</Button>
            <Button onClick={handleCadastrarNovo} disabled={salvando || !novoNome.trim()}>
              {salvando ? 'Cadastrando...' : 'Cadastrar'}
            </Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </>
  )
}
```

```typescript
  const [open, setOpen] = useState(false)
  const [clienteId, setClienteId] = useState<string | null>(null)
  const [clienteSelecionado, setClienteSelecionado] = useState<{ id: string; nome: string } | null>(null)
```
